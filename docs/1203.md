WITH top5 AS (

        f.news_id,
        ROW_NUMBER() OVER (ORDER BY f.publish_time ASC) AS rn
    FROM feed_item f
    WHERE f.is_top_official = TRUE
    ORDER BY f.publish_time ASC
    LIMIT 5
),
normal15 AS (
    SELECT 
        f.news_id,
        ROW_NUMBER() OVER (ORDER BY f.publish_time ASC) AS rn
    FROM feed_item f
    WHERE f.is_top_official = FALSE
    ORDER BY f.publish_time ASC
    LIMIT 15
)
UPDATE feed_item AS f
SET publish_time =
    CASE 
        WHEN f.news_id IN (SELECT news_id FROM top5)
            THEN NOW()                                  -- Top5 = 最新时间
        ELSE 
            NOW() + (n.rn * INTERVAL '1 second')        -- Normal15 = 递增时间
    END
FROM normal15 n
WHERE f.news_id IN (
    SELECT news_id FROM top5
    UNION ALL
    SELECT news_id FROM normal15
);
